Ветвь: t0.infra.registry
Слой: T0
Методов: 24
Генерируется tools/registry_scan.py; вручную не редактировать.
Описание: Генератор реестра методов: индекс registry/METHODS.txt и
          карточки по исходным файлам registry/<ветвь>/<файл>.txt. Режим
          --check сверяет реестр без записи.

МЕТОД _strip_tag(line) -> tuple
  вход:        line: str — строка исходного файла
  выход:       pair: tuple — (ключ, значение) для строки "# @ключ
               значение", иначе (None, "")
  функция:     Распознаёт строки-аннотации "# @ключ значение".

МЕТОД parse_file_header(lines) -> dict
  вход:        lines: list[str] — строки файла
  выход:       header: dict — поля file, layer, branch, desc (список
               строк)
  функция:     Читает шапку файла: непрерывный блок "# @..." в начале
               файла.

МЕТОД parse_method_cards(lines) -> list
  вход:        lines: list[str] — строки файла
  выход:       cards: list[dict] — карточки с полями name, in, out,
               does, uses, depends
  функция:     Собирает карточки: строка "@method" открывает карточку,
               последующие строки "@in/@out/@does/@uses/@depends"
               наполняют её, любая другая строка закрывает.

               .py-файлы.

  выход:       entries: list[dict] — элементы: path, header, cards

МЕТОД wrap_words(text, width) -> list

  функция:     Переносит текст по словам; слово длиннее width занимает

МЕТОД merge_items(key, items) -> list
  вход:        key: str — ключ поля (in, out, does, uses, depends)
  вход:        items: list[str] — значения поля, как в исходнике
  выход:       merged: list[str] — абзацы поля после склейки
  функция:     Склеивает продолжения: @does — в один текст через пробел;
               @uses и @depends — в один список через запятую; @in и
               @out остаются отдельными строками.
