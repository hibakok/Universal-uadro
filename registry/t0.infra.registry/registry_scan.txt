Ветвь: t0.infra.registry
Слой: T0
Методов: 14
Генерируется tools/registry_scan.py; вручную не редактировать.
Описание: Сканер реестра методов: чтение исходников, план (ветви и имена
          карточек), запись и сверка registry/ на диске; рендер текстов
          — tools/registry_render.py. Режим --check сверяет реестр без
          записи.

МЕТОД _strip_tag(line) -> tuple
  вход:        line: str — строка исходного файла
  выход:       pair: tuple — (ключ, значение) для строки "# @ключ
               значение"
  выход:       (Python) или "; @ключ значение" (uAsm/uBrick, Этап 4),
  выход:       иначе (None, "")
  функция:     Распознаёт строки-аннотации обоих видов: после маркера
               комментария (# или ;) идёт пробел и "@ключ значение"
               (SPEC/L4-SELFHOST.txt, раздел 3).

МЕТОД parse_file_header(lines) -> dict
  вход:        lines: list[str] — строки файла
  выход:       header: dict — поля file, layer, branch, desc (список
               строк)
  функция:     Читает шапку файла: непрерывный блок "# @..." в начале
               файла.

МЕТОД parse_method_cards(lines) -> list
  вход:        lines: list[str] — строки файла
  выход:       cards: list[dict] — карточки с полями name, in, out,
               does, uses, depends
  функция:     Собирает карточки: строка "@method" открывает карточку,
               последующие строки "@in/@out/@does/@uses/@depends"
               наполняют её, любая другая строка закрывает.

МЕТОД collect_sources() -> list
  вход:        нет
  выход:       paths: list[Path] — отсортированные исходники расширений
  выход:       SOURCE_EXTS (*.py, *.uasm, *.ubrk) из каталогов
               исходников
  функция:     Обходит существующие каталоги SOURCE_DIRS и собирает
               файлы расширений SOURCE_EXTS: Python-инструменты и
               исходники собственных языков (SPEC/L4-SELFHOST.txt,
               раздел 3).

МЕТОД read_entries() -> list
  вход:        нет
  выход:       entries: list[dict] — элементы: path, header, cards
  функция:     Читает все исходники и разбирает шапки и карточки.
  зависимости: collect_sources, parse_file_header, parse_method_cards

МЕТОД branch_of(entry) -> str
  вход:        entry: dict — элемент реестра (path, header, cards)
  выход:       branch: str — имя ветви или no-branch, если в шапке нет
               ветви
  функция:     Достаёт имя ветви из шапки файла.

МЕТОД rel_path(entry) -> str
  вход:        entry: dict — элемент реестра (path, header, cards)
  выход:       rel: str — путь файла относительно корня репозитория
  функция:     Преобразует путь в POSIX-вид для плана реестра.

МЕТОД card_name_for(entry, used) -> str
  вход:        entry: dict — элемент реестра (path, header, cards)
  вход:        used: set[str] — уже занятые имена карточек этой ветви
  выход:       name: str — имя файла карточек, уникальное внутри ветви
  функция:     Строит имя карточек из имени исходника; при совпадении
               имён внутри ветви добавляет числовой суффикс. Имена
               сравниваются без учёта регистра (файловые системы узла
               могут не различать регистр); имя индекса ветви
               (INDEX.txt) занято и карточкам не выдаётся.

МЕТОД plan_registry(entries) -> list
  вход:        entries: list[dict] — элементы реестра
  выход:       plan: list[dict] — элементы: branch, entry, card (имя
               файла
  выход:       карточек), rel (путь исходника)
  функция:     Назначает каждому исходнику его ветвь, имя файла карточек

               каждой ветви (BRANCH_INDEX_NAME).
  зависимости: branch_of, rel_path, card_name_for
